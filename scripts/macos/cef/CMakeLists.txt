cmake_minimum_required(VERSION 3.19)
project(GamePortalCEF LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
add_library(portal_cef STATIC ../cef-bridge.mm)
target_include_directories(portal_cef PRIVATE "${CEF_ROOT}")
target_compile_definitions(portal_cef PRIVATE CEF_USE_SANDBOX)
set_target_properties(portal_cef PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_executable(portal_cef_helper ../cef-bridge.mm)
target_include_directories(portal_cef_helper PRIVATE "${CEF_ROOT}")
target_compile_definitions(portal_cef_helper PRIVATE CEF_USE_SANDBOX PORTAL_CEF_HELPER)
target_link_libraries(portal_cef_helper libcef_dll_wrapper "${CEF_SANDBOX_LIB_RELEASE}" ${CEF_STANDARD_LIBS})
set_target_properties(portal_cef_helper PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
